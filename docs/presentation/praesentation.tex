\section{Projektkontext}

\begin{frame}{Projektziel}
  \begin{block}{Zielbild}
    PMA zeigt Social Discovery nicht nur ueber Personen-Matching, sondern ueber einen Produktfluss aus Profil, Discovery, Match, Chat, swipebaren Kontexten und Karte.
  \end{block}

  \vspace{0.3cm}

  \begin{itemize}
    \item technisch vorfuehrbarer End-to-End-MVP
    \item bewusster MVP statt Produktivsystem
    \item Umsetzung in vier Sprints
  \end{itemize}
\end{frame}

\begin{frame}{SMART-Ziel}
  \begin{itemize}
    \item \textbf{S:} Profil, Swipe, Match, Chat, Events und Karte in einem lauffaehigen MVP
    \item \textbf{M:} alle Kernflows sind in der Demo benutzbar
    \item \textbf{A:} breiter als reines Personen-Matching
    \item \textbf{R:} ohne Live-Chat, echten Upload und tiefe Empfehlungslogik
    \item \textbf{T:} Umsetzung in vier Sprints
  \end{itemize}
\end{frame}

\begin{frame}{Warum ist PMA ein Projekt?}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Projektcharakter}
    \begin{itemize}
      \item neues Produktziel
      \item begrenzter Zeitrahmen
      \item Rollen und Zuständigkeiten
      \item Integrations- und Scope-Risiken
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Nicht Routine}
    \begin{itemize}
      \item kein Standardprozess
      \item Produktbild entstand iterativ
      \item Anforderungen aenderten sich durch Reviews
      \item Frontend, Backend und Datenmodell wuchsen gemeinsam
    \end{itemize}
  \end{columns}
\end{frame}

\section{Produkt}

\begin{frame}{Produktidee}
  \begin{itemize}
    \item PMA ist keine reine Dating-App, sondern ein Social-Discovery-MVP.
    \item Menschen bleiben der zentrale Flow fuer Match und Chat.
    \item Discovery wird um Hobbys, Events und Locations erweitert.
  \end{itemize}

  \vspace{0.3cm}

  \begin{block}{Kernfrage}
    Wie laesst sich Kennenlernen digital ueber Personen, gemeinsame Interessen, Events und Orte denken?
  \end{block}
\end{frame}

\begin{frame}{Umgesetzter MVP-Umfang}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Nutzerfluss}
    \begin{itemize}
      \item Registrierung und Login
      \item Profilpflege mit Galerie
      \item Swipe fuer Menschen, Hobbys, Events und Locations
      \item Match-Logik fuer Menschen
      \item Chat zwischen Matches
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Sozialer Kontext}
    \begin{itemize}
      \item Events erstellen, filtern und bearbeiten
      \item Kartenansicht mit Event-Markern
      \item pseudoanonymisierte Nutzerzonen
      \item Orts- und Stadtvorschlaege ueber Nominatim/OSM \cite{nominatim}
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Technischer Aufbau}
  \begin{tabularx}{\textwidth}{@{}>{\bfseries}l X@{}}
    Frontend & React, React Router, Tailwind, React Leaflet \\
    Backend & Node.js, Express, MongoDB, Mongoose \\
    Auth & tokenbasierte Session-Authentifizierung \\
    Tests & \texttt{node --test} mit \texttt{mongodb-memory-server} \\
    Demo & Seed-Script fuer reproduzierbare Datenlage \\
  \end{tabularx}

  \vspace{0.4cm}

  \begin{block}{Architekturidee}
    Statt isolierter Screens wurde ab Sprint 2 bewusst auf vertikale Inkremente gesetzt: UI, API und Datenmodell pro Kernfunktion gemeinsam.
  \end{block}
\end{frame}

\begin{frame}{Architektur auf einen Blick}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Frontend}
    \begin{itemize}
      \item React SPA mit geschuetzten Routen
      \item Swipe, Matches, Chat, Events, Karte
      \item REST-Kommunikation
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Backend}
    \begin{itemize}
      \item Express-API unter \texttt{/api/v1}
      \item Module fuer Auth, Discovery, Matches, Chat, Events, Map
      \item MongoDB als zentrale Persistenz
    \end{itemize}
  \end{columns}

  \vspace{0.3cm}

  \begin{block}{Wichtige Entscheidung}
    Monolithisches REST-Backend: weniger Integrationsaufwand, schneller demo-faehig.
  \end{block}
\end{frame}

\section{Sprintverlauf}

\begin{frame}{Sprintverlauf in Kurzform}
  \begin{enumerate}
    \item \textbf{Sprint 1:} klickbares Produktbild, aber noch stark mockup-lastig
    \item \textbf{Sprint 2:} echter Kernflow mit Auth, Profil, Discovery, Swipe und Match
    \item \textbf{Sprint 3:} Match-Nutzen durch Chat, dazu erste Event- und Kartenbereiche
    \item \textbf{Sprint 4:} Backend-Anbindung fuer Events und Karte, Multi-Deck-Discovery, Demo-Seed, Doku und Stabilisierung
  \end{enumerate}
\end{frame}

\begin{frame}{Wichtigstes Review-Learning}
  \begin{block}{Rueckmeldung nach Sprint 1}
    Ein reiner Frontend-Prototyp reicht nicht aus. PMA musste frueh als echter End-to-End-Flow erfahrbar werden.
  \end{block}

  \begin{itemize}
    \item Wechsel von Mockup-Denken zu vertikaler Umsetzung
    \item Personen-Discovery blieb Kern
    \item Stakeholder-Wunsch nach Hobbys, Events und Locations floss sichtbar in den spaeteren Scope ein
  \end{itemize}
\end{frame}

\begin{frame}{Zentrale User Stories}
  \begin{itemize}
    \item Als neuer Nutzer will ich mich registrieren und einloggen koennen, damit ich PMA nutzen kann.
    \item Als Nutzer will ich mein Profil pflegen koennen, damit andere mich sinnvoll entdecken koennen.
    \item Als Nutzer will ich Menschen, Hobbys, Events und Locations swipen koennen, damit Discovery breiter als Dating gedacht ist.
    \item Als Match will ich chatten koennen, damit aus einem Match echte Interaktion wird.
    \item Als Nutzer will ich Events ansehen, filtern und auf einer Karte einordnen koennen, damit spontane Treffen sichtbar werden.
  \end{itemize}
\end{frame}

\section{Einordnung}

\begin{frame}{Bewusste MVP-Grenzen}
  \begin{itemize}
    \item Bilder werden aktuell clientseitig als komprimierte Data-URLs gespeichert
    \item kein Echtzeit-Chat per WebSocket
    \item Nicht-Personen-Swipes haben noch keine persistente Logik oder Matchmechanik
    \item Kartenzonen sind fuer die Demo bewusst pseudoanonymisiert
  \end{itemize}
\end{frame}

\begin{frame}{Fazit}
  \begin{itemize}
    \item PMA zeigt einen technisch durchgaengigen Social-Discovery-MVP.
    \item Der groesste methodische Fortschritt war der Wechsel von klickbaren Screens zu vertikalen Inkrementen.
    \item Scrum war fuer das Projekt vor allem als Struktur fuer Priorisierung, Review-Feedback und Steuerung nuetzlich.
  \end{itemize}

  \vspace{0.3cm}

  \begin{block}{Kurz gesagt}
    Aus einer fruehen UI-Idee wurde ein vorfuehrbares Produkt mit nachvollziehbarem Sprintverlauf und klaren MVP-Grenzen.
  \end{block}
\end{frame}
